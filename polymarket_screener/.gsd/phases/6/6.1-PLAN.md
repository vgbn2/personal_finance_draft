    Create a `BaseStrategy` abstract class that defines `on_market_update` and `on_signal`. 
    Implement a `StrategyRegistry` that dynamically discovers and loads classes inheriting from `BaseStrategy` inside the `app/strategies/` folder. This allows drop-in deployment of new algorithms without touching core engine code.
  </action>
  <verify>python -c "import app.core.strategy_registry"</verify>
  <done>Registry successfully detects and loads dummy strategies from the filesystem.</done>
</task>

```python
# CODE SKETCH: Strategy Plugin System
import importlib
import os

class StrategyRegistry:
    def __init__(self):
        self.strategies = {}

    def load_plugins(self, directory: str = "app/strategies"):
        for filename in os.listdir(directory):
            if filename.endswith(".py") and filename != "base.py":
                module_name = filename[:-3]
                module = importlib.import_module(f"app.strategies.{module_name}")
                # Instantiate any class inheriting from BaseStrategy
                for attr_name in dir(module):
                    attr = getattr(module, attr_name)
                    if isinstance(attr, type) and issubclass(attr, BaseStrategy):
                        self.strategies[attr.name] = attr()
```

    master_circuit_breaker.trip("API Kill Switch Activated via Frontend")

from app.api.rest_endpoints import router as api_router

app = FastAPI()
app.include_router(ws_router)
app.include_router(api_router)

async def main():
    # Start engine tasks
    tasks = [
        event_bus.start(),
        execution_router.listen(),
        screener.listen(),
    ]
    
    # Auto-detect port availability
    import socket
    s = socket.socket(socket.socket(socket.AF_INET, socket.SOCK_STREAM))
    try:
        s.bind(("0.0.0.0", 8000))
        s.close()
        # Port is free, start Uvicorn...
    except socket.error:
        log.error("Port 8000 is busy. Is another instance of the bot running?")
        return

    await asyncio.gather(*tasks)
