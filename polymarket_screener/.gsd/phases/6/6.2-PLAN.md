```python
# CODE SKETCH: Async Audit Stream Daemon
class AuditDaemon:
    def __init__(self, event_queue: asyncio.Queue, db_manager: DatabaseManager):
        self.queue = event_queue
        self.db = db_manager
        
    async def listen(self):
        batch = []
        while True:
            # Gather events silently in background
            event = await self.queue.get()
            batch.append(event)
            
            # Flush to DB every 10 events or timeout
            if len(batch) >= 10:
                await self.db.insert_audit_logs_many(batch)
                batch.clear()
```
