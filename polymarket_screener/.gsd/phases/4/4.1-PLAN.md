# CODE SKETCH: UI Adapter (app/api/ui_adapter.py)
class UIAdapter:
    @staticmethod
    def format_market_state(backend_state: dict) -> dict:
        return {
            "id": backend_state["market_id"],
            "title": backend_state["title"],
            "fairValue": round(backend_state["p_real"] * 100, 1),
            "marketPrice": round(backend_state["best_ask"] * 100, 1),
            "edge": round(backend_state["edge"] * 100, 2)
        }
```

# CODE SKETCH: WebSocket State Pushing (app/api/ws_bridge.py)
from fastapi import WebSocket, APIRouter
import asyncio

router = APIRouter()

@router.websocket("/ws/state")
async def websocket_endpoint(websocket: WebSocket):
    await websocket.accept()
    while True:
        state = UIAdapter.format_market_state(system_state.get_current())
        await websocket.send_json({"type": "STATE_SYNC", "data": state})
        await asyncio.sleep(0.5)
```
