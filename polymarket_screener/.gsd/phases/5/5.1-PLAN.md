from motor.motor_asyncio import AsyncIOMotorClient

class DatabaseManager:
    def __init__(self, uri: str):
        self.client = AsyncIOMotorClient(uri, serverSelectionTimeoutMS=5000)
        self.db = self.client.poly_screener

    async def insert_audit_log(self, data: dict):
        try:
            await self.db.audit_logs.insert_one(data)
        except Exception as e:
            log.error(f"DB Insert Failed (Non-Blocking): {e}")
```

class ExecutionRouter:
    def __init__(self, event_queue: asyncio.Queue):
        self.queue = event_queue
    
    async def listen(self):
        while True:
            signal = await self.queue.get()
            gate = risk_engine.check_gates(signal)
            if gate.status == "PASSED":
                await self.send_order(signal)
            else:
                log.warning(f"Router rejected signal: {gate.rejection_reason}")
```
