```python
from hypothesis import given, strategies as st
import numpy as np

@given(
    S=st.floats(min_value=0.01, max_value=100.0),
    K=st.floats(min_value=0.01, max_value=100.0),
    T=st.floats(min_value=0.0, max_value=1.0), # Includes T=0 edge case
    r=st.floats(min_value=-0.1, max_value=0.1),
    sigma=st.floats(min_value=0.0, max_value=5.0) # Includes Vol=0 edge case
)
def test_black_scholes_never_nans(S, K, T, r, sigma):
    from app.math.black_scholes import vectorized_nd2
    result = vectorized_nd2(np.array([S]), np.array([K]), np.array([T]), r, np.array([sigma]))
    assert not np.isnan(result).any()
```
