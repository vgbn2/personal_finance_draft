class MarketScreener:
    async def process_snapshot(self, snapshot: dict):
        market_prob = snapshot['yes_price']
        fair_prob = pricing.calculate_fair_value(snapshot['orderbook'])
        edge = fair_prob - market_prob
        
        if edge > config.min_edge:
            await event_bus.publish("SIGNAL_DETECTED", {"market": snapshot['id'], "edge": edge})
```

# CODE SKETCH: Portfolio Tracker (app/core/portfolio.py)
class PortfolioManager:
    def __init__(self):
        self.positions = {}
        
    def calculate_mtm_pnl(self, current_prices: dict) -> float:
        total_pnl = 0.0
        for market_id, pos in self.positions.items():
            current_price = current_prices.get(market_id, pos.entry_price)
            # Mark to market value
            total_pnl += (current_price - pos.entry_price) * pos.size
        return total_pnl
```
