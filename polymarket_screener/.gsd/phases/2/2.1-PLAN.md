# CODE SKETCH: Vectorized Black-Scholes (app/math/black_scholes.py)
import numpy as np
from scipy.stats import norm

def vectorized_nd2(S: np.ndarray, K: np.ndarray, T: np.ndarray, r: float, sigma: np.ndarray) -> np.ndarray:
    d1 = (np.log(S / K) + (r + 0.5 * sigma ** 2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    return norm.cdf(d2)
```

# CODE SKETCH: Kelly Allocation (app/math/kelly.py)
def calculate_kelly(win_prob: float, odds_offered: float) -> float:
    # f* = (bp - q) / b
    b = odds_offered - 1.0  # Decimal odds - 1
    q = 1.0 - win_prob
    target_fraction = (b * win_prob - q) / b
    return max(0.0, target_fraction)
```
