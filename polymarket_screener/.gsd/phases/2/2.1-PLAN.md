```python
# CODE SKETCH: Market Screener logic
class MarketScreener:
    def filter_by_asset(self, events, asset_type="Crypto"):
        return [e for e in events if e.category == asset_type]
    def rank_by_liquidity(self, events):
        return sorted(events, key=lambda x: x.volume, reverse=True)
```
