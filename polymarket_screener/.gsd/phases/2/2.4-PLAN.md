```python
# CODE SKETCH: 3D Risk Gates (app/execution/risk.py)
class RiskManager:
    def check_gates(self, signal: dict, portfolio: PortfolioManager) -> tuple[bool, str]:
        if signal['liquidity'] < config.min_liquidity:
            return False, "Failed Liquidity Gate"
        if portfolio.exposure + signal['size'] > config.max_exposure:
            return False, "Failed Exposure Gate"
        return True, "Passed"
```
