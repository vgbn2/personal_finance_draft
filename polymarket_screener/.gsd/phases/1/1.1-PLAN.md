```python
def setup_dirs(base_path):
    dirs = ['config', 'data/streams', 'data/clients', 'strategy', 'api', 'frontend']
    for d in dirs:
        os.makedirs(os.path.join(base_path, d), exist_ok=True)
```

```python
class SymbolRegistry:
    def load(self):
        with open('config/symbols.yaml') as f:
            self.map = yaml.safe_load(f)
    def resolve(self, label):
        return self.map.get(label) # returns {'binance': 'BTCUSDT', 'polymarket': '...', 'type': 'crypto'}
```
