```python
from pydantic import BaseModel, Field
from typing import List, Tuple

class UnifiedOrderbook(BaseModel):
    exchange: str
    symbol: str
    bids: List[Tuple[float, float]] # [price, amount]
    asks: List[Tuple[float, float]]
    timestamp: float = Field(default_factory=time.time)

    @classmethod
    def from_polymarket(cls, raw: dict):
        # Translate Polymarket specific field names
        return cls(exchange="polymarket", bids=raw['b'], asks=raw['a'], symbol=raw['id'])
```
