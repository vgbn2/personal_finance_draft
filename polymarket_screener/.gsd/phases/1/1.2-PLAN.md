```python
class WSBridge:
    async def handle_ws(self, websocket):
        await websocket.accept()
        try:
            while True:
                data = await self.queue.get()
                await websocket.send_json(data)
        except WebSocketDisconnect:
            pass # Graceful cleanup on VPN drop
```
